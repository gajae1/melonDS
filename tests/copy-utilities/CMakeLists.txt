cmake_minimum_required(VERSION 3.30)
project(melonDSCopyUtilities LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
enable_testing()
add_executable(CopyUtilities CopyUtilities.cpp ../../src/Utils.cpp)
target_include_directories(CopyUtilities PRIVATE ../../src)
add_test(NAME copy-utilities COMMAND CopyUtilities)

add_executable(BufferUtilities ../BufferUtilities.cpp ../../src/Utils.cpp)
target_include_directories(BufferUtilities PRIVATE ../../src)
add_test(NAME rom-buffer-copies COMMAND BufferUtilities copies)
add_test(NAME rom-buffer-overflow COMMAND BufferUtilities overflow)
add_test(NAME rom-buffer-null COMMAND BufferUtilities null)
set_tests_properties(rom-buffer-overflow rom-buffer-null PROPERTIES TIMEOUT 5)
